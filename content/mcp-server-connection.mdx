## 🔍 Understanding stdio Transport

### What is stdio Transport?
- **No Network Port**: Communication happens via stdin/stdout, not network sockets
- **Process-based**: Client spawns the server process as a subprocess
- **JSON-RPC**: Messages are exchanged using JSON-RPC protocol over standard I/O
- **Secure**: No network exposure, communication is contained within the process

### Benefits
- **Security**: No network ports exposed
- **Simplicity**: Easy to set up and configure
- **Performance**: Direct process communication
- **Isolation**: Each client gets its own server instance

### Limitations
- **Local Only**: Cannot be accessed from remote machines
- **Process Management**: Client must handle server process lifecycle
- **Single Client**: Each server instance handles one client

## 🛠️ Client Implementation Examples

### Python Client
```python
import subprocess
import json
import asyncio

class MCPClient:
    def __init__(self):
        self.process = None
    
    async def start_server(self):
        """Start the MCP server process"""
        self.process = subprocess.Popen(
            ["python3", "mcp_tools/simple_rag_generator.py"],
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            cwd="/home/muthu/projects/AI/CascadeProjects/windsurf-project"
        )
    
    async def call_tool(self, tool_name, params):
        """Call a tool on the MCP server"""
        request = {
            "jsonrpc": "2.0",
            "method": "tools/call",
            "params": {
                "name": tool_name,
                "arguments": params
            },
            "id": 1
        }
        
        # Send request
        self.process.stdin.write(json.dumps(request) + "\n")
        self.process.stdin.flush()
        
        # Read response
        response_line = self.process.stdout.readline()
        response = json.loads(response_line.strip())
        
        return response.get("result", {})
    
    async def close(self):
        """Close the server process"""
        if self.process:
            self.process.terminate()
            await self.process.wait()

# Usage
async def main():
    client = MCPClient()
    await client.start_server()
    
    try:
        # Generate a component
        result = await client.call_tool("generate_component_from_query", {
            "query": "Create a primary button with hover effects",
            "framework": "React",
            "style_mode": "css-module"
        })
        
        print("Generated Component:")
        print(result["generated_code"]["component"])
        
    finally:
        await client.close()

if __name__ == "__main__":
    asyncio.run(main())
```

### Node.js Client
```javascript
const { spawn } = require('child_process');

class MCPClient {
  constructor() {
    this.server = null;
    this.requestId = 1;
    this.pendingRequests = new Map();
  }
  
  startServer() {
    return new Promise((resolve, reject) => {
      this.server = spawn('python3', [
        '/home/muthu/projects/AI/CascadeProjects/windsurf-project/mcp_tools/simple_rag_generator.py'
      ], {
        cwd: '/home/muthu/projects/AI/CascadeProjects/windsurf-project',
        stdio: ['pipe', 'pipe', 'inherit']
      });
      
      this.server.stdout.on('data', (data) => {
        const lines = data.toString().trim().split('\n');
        lines.forEach(line => {
          if (line.trim()) {
            try {
              const response = JSON.parse(line);
              const pending = this.pendingRequests.get(response.id);
              if (pending) {
                pending.resolve(response.result || response.error);
                this.pendingRequests.delete(response.id);
              }
            } catch (e) {
              console.error('Failed to parse response:', line);
            }
          }
        });
      });
      
      this.server.on('error', reject);
      this.server.on('spawn', resolve);
    });
  }
  
  async callTool(toolName, params) {
    return new Promise((resolve, reject) => {
      const request = {
        jsonrpc: "2.0",
        method: "tools/call",
        params: {
          name: toolName,
          arguments: params
        },
        id: this.requestId++
      };
      
      this.pendingRequests.set(request.id, { resolve, reject });
      
      this.server.stdin.write(JSON.stringify(request) + '\n');
      
      // Timeout after 30 seconds
      setTimeout(() => {
        const pending = this.pendingRequests.get(request.id);
        if (pending) {
          this.pendingRequests.delete(request.id);
          reject(new Error('Request timeout'));
        }
      }, 30000);
    });
  }
  
  close() {
    if (this.server) {
      this.server.kill();
    }
  }
}

// Usage
async function main() {
  const client = new MCPClient();
  
  try {
    await client.startServer();
    
    // Generate a component
    const result = await client.callTool('generate_component_from_query', {
      query: 'Create a primary button with hover effects',
      framework: 'React',
      style_mode: 'css-module'
    });
    
    console.log('Generated Component:');
    console.log(result.generated_code.component);
    
  } catch (error) {
    console.error('Error:', error);
  } finally {
    client.close();
  }
}

main().catch(console.error);
```
